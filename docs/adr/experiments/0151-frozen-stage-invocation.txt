def check(ok: bool = True, message: str = 'arithmetic check') -> None:
    if not ok:
        raise ValueError(message)
    if not 0 <= monotonic()-START < 5:
        raise ValueError('five-second offline budget')


def load(path: Path) -> Any:
    def reject(value: str) -> None:
        raise ValueError('nonfinite JSON '+value)
    return json.loads(path.read_text(),parse_constant=reject)


def force(delta: tuple[Decimal, ...], mu: Decimal) -> tuple[Decimal, ...]:
    global FIELDS
    check(FIELDS < 234, 'Decimal field cap')
    FIELDS += 1
    r2 = sum(v*v for v in delta)
    check(r2 > 0, 'positive separation')
    return tuple(mu*v/(r2*r2.sqrt()) for v in delta)


def estimate(stages: list[tuple[Decimal, ...]], e3: list[float], e5: list[float], scale: list[float]) -> dict[str, Any]:
    d = Decimal.from_float
    u = Decimal(2)**-53
    gamma = 32*u/(1-32*u)
    vectors, bounds = {}, {}
    for name, coeff in (('E3',e3),('E5',e5)):
        raw = [sum(stages[i][j]*d(coeff[i]) for i in range(13))/d(scale[j]) for j in range(6)]
        bound = [gamma*sum(abs(stages[i][j]*d(coeff[i])) for i in range(13))/d(scale[j])+u*abs(raw[j])+Decimal('1e-40') for j in range(6)]
        vectors[name], bounds[name] = raw,bound
    a = sum(v*v for v in vectors['E5'])
    b = sum(v*v for v in vectors['E3'])
    norm = Decimal(0) if a == b == 0 else a/(6*(a+Decimal('0.01')*b)).sqrt()
    lower5 = sum(max(Decimal(0),abs(x)-z)**2 for x,z in zip(vectors['E5'],bounds['E5']))
    upper5 = sum((abs(x)+z)**2 for x,z in zip(vectors['E5'],bounds['E5']))
    lower3 = sum(max(Decimal(0),abs(x)-z)**2 for x,z in zip(vectors['E3'],bounds['E3']))
    upper3 = sum((abs(x)+z)**2 for x,z in zip(vectors['E3'],bounds['E3']))
    low = Decimal(0) if lower5 == 0 else max(Decimal(0),lower5/(6*(lower5+Decimal('0.01')*upper3)).sqrt()-Decimal('1e-40'))
    high = Decimal(0) if upper5 == 0 else upper5/(6*(upper5+Decimal('0.01')*lower3)).sqrt()+Decimal('1e-40')
    return {'E3':vectors['E3'],'E5':vectors['E5'],'E3_arithmetic_bound':bounds['E3'],'E5_arithmetic_bound':bounds['E5'],'norm':norm,'norm_box':[low,high],'box_classification':'below_one' if high < 1 else 'above_one' if low > 1 else 'ambiguous'}


def wire(value: Any) -> Any:
    if isinstance(value,Decimal):
        check(value.is_finite(), 'finite Decimal output')
        return str(value)
    if isinstance(value,dict):
        return {k:wire(v) for k,v in value.items()}
    if isinstance(value,(list,tuple)):
        return [wire(v) for v in value]
    return value
