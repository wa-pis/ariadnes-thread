def load(mode: str, expected: str) -> dict[str, Any]:
    raw = gzip.decompress((HERE/f'0164-corrected-{mode}.json.gz').read_bytes())
    assert sha256(raw).hexdigest() == expected, mode
    return json.loads(raw)


def frac(wire: dict[str, str]) -> Fraction:
    return F(int(wire['numerator']), int(wire['denominator']))


def norm_um(vector: list[Fraction]) -> float:
    return math.hypot(*(float(x*1_000_000) for x in vector))


def dec(value: Fraction) -> Decimal:
    return D(value.numerator)/D(value.denominator)


def diagnostic_error(geometry: list[Fraction], gm: Fraction, ref: list[str],
                     precision: int) -> Decimal:
    with localcontext() as ctx:
        ctx.prec = precision
        d = list(map(dec,geometry))
        rho2 = sum(x*x for x in d)
        factor = dec(gm)/(rho2*rho2.sqrt())
        return sum((factor*x-D(y))**2 for x,y in zip(d,ref,strict=True)).sqrt()
