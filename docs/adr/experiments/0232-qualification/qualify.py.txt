"""R5.59t: exact command dispatch through test-only artificial dependencies."""
from hashlib import sha256
import json
from pathlib import Path
import shutil
import sys
from time import monotonic_ns

ROOT = Path('/Users/agrudin/dev/my/ariadna')
BASE = Path('/private/tmp/ariadna-dispatch-repair-20261121')
sys.path.insert(0, str(ROOT / 'tests'))
from test_lineage_dispatch import (
    SCRIPT, _assert_dispatch, _inventory_bytes, invoke, load_dispatcher,
    prepare_dispatch_case, telemetry,
)

def wire(value):
    return (json.dumps(value, indent=2, sort_keys=True, allow_nan=False) + '\n').encode()

case = prepare_dispatch_case(BASE / 'artificial-full')
records = []
for role in ('producer', 'auditor'):
    if role == 'producer':
        args = ['retained']
    else:
        args = ['audit', sha256((case['output'] / 'completion.json').read_bytes()).hexdigest()]
    start = monotonic_ns()
    result = invoke(case, *args)
    elapsed = monotonic_ns() - start
    (BASE / f'{role}.stdout').write_text(result.stdout)
    (BASE / f'{role}.stderr').write_text(result.stderr)
    assert result.returncode == 0, result.stderr
    _assert_dispatch(case, result, args)
    rows = telemetry(result)
    records.append(dict(role=role, args=args, returncode=result.returncode,
                        external_elapsed_ns=elapsed, telemetry=rows))

report = json.loads((case['output'] / 'report.json').read_bytes())
audit = json.loads(case['audit'].read_bytes())
assert report['scope'] == 'artificial-saved-lineage'
assert report['status'] == 'lineage-qualified-attribution-unresolved'
assert audit['status'] == 'lineage-independently-verified'
assert report['source_inventory_bytes'] == _inventory_bytes(case) <= 4 * 1024**2
assert report['source_pins'] == case['profile']['pins']
assert report['commands'] is None
assert report['physical_status'] == 'physical-join-not-established'
assert report['work'] == audit['work'] == audit['proof']['counts']
output_bytes = sum(path.stat().st_size for path in case['output'].iterdir())
assert output_bytes <= 1048576 - 65536
assert case['audit'].stat().st_size <= 65536
combined = output_bytes + case['audit'].stat().st_size
assert combined <= 1048576
module = load_dispatcher()
contract = json.loads(module.CONTRACT.read_bytes())
paths = {case['bootstrap'], case['config'], module.CONTRACT}
paths.update(Path(path) for path in case['profile']['inputs'].values())
paths.update(SCRIPT.with_name(name) for name in module.PRIVATE)
paths.update(ROOT / path for path in contract['source_bindings'].values())
inventory = {str(path): dict(bytes=path.stat().st_size, sha256=sha256(path.read_bytes()).hexdigest())
             for path in sorted(paths)}
assert sum(row['bytes'] for row in inventory.values()) == _inventory_bytes(case)
source_paths = [SCRIPT, ROOT / 'tests/lineage_dispatch_harness.py', ROOT / 'tests/test_lineage_dispatch.py']
summary = dict(stage='R5.59t', scope='artificial-saved-lineage',
               inherited_raw_stage=report['stage'], report_status=report['status'],
               audit_status=audit['status'], work=report['work'], measurements=records,
               source_inventory_bytes_per_process=report['source_inventory_bytes'],
               source_inventory=inventory, producer_output_bytes=output_bytes,
               auditor_output_bytes=case['audit'].stat().st_size, combined_output_bytes=combined,
               measured_source_sha256={str(path.relative_to(ROOT)):sha256(path.read_bytes()).hexdigest()
                                       for path in source_paths},
               real_retained_input_reads=0, r559s_repeated=False, spice_queries=0,
               integrations=0, r560_started=False,
               fixture_recipe='tests/lineage_fixtures.py:write_artificial_pack; full unpadded fixture',
               source_redirection='test-only bootstrap; exact argv; real admission/authentication/guards/analyzer/verifier')
(BASE / 'artificial-summary.json').write_bytes(wire(summary))
(BASE / 'artificial-profile.json').write_bytes(wire(case['profile']))
print(json.dumps({key: value for key, value in summary.items()
                  if key not in {'measurements', 'source_inventory'}}))
print(json.dumps([dict(role=row['role'], external_elapsed_ns=row['external_elapsed_ns'],
                       returncode=row['returncode']) for row in records]))
