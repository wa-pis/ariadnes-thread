def write(path, data):
    raw = data if isinstance(data, bytes) else (json.dumps(data, sort_keys=True, indent=2, allow_nan=False) + '\n').encode()
    with path.open('xb') as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())
    handle = os.open(path.parent, os.O_RDONLY)
    try:
        os.fsync(handle)
    finally:
        os.close(handle)

commands = [sys.executable, '-I', '-S', '-B', str(SCRIPT)]

def dispatch(name, args):
    write(E / (name + '-attempt.json'), dict(command=commands + args,
          utc=datetime.now(timezone.utc).isoformat(), expected_wall_ns=30_000_000_000, ordinal=1))
    started = monotonic_ns()
    try:
        child = subprocess.run(commands + args, stdin=subprocess.DEVNULL, capture_output=True, timeout=31, check=False)
        stdout, stderr = child.stdout, child.stderr
        result = dict(returncode=child.returncode, timed_out=False)
    except subprocess.TimeoutExpired as error:
        stdout, stderr = error.stdout or b'', error.stderr or b''
        result = dict(returncode=None, timed_out=True)
    except OSError as error:
        stdout, stderr = b'', str(error).encode()
        result = dict(returncode=None, timed_out=False, launch_error=str(error))
    result |= dict(external_elapsed_ns=monotonic_ns()-started, command=commands+args, process=name)
    write(E / (name + '.stdout'), stdout)
    write(E / (name + '.stderr'), stderr)
    write(E / (name + '-measurement.json'), result)
    return result
